import cv2

def target_inference(frame):
    # 初始化 YOLO 模型
    model = torch.hub.load('ultralytics/yolov8', 'custom', path='weights/best_uav.pt')
    
    # 执行推理
    results = model(frame)
    
    # 提取无人机坐标
    detections = results.xyxy[0] 
    
    # 过滤置信度 < 0.8 的目标
    valid_targets = [d for d in detections if d[4] > 0.8]
    
    return valid_targets, results.render()[0]
